\begin{preguntas}

% Ejercicio: electrostatica-11-046
\pregunta[1] Dos esferas se atraen con una fuerza determinada. ¿Cómo cambia el valor de la
fuerza si se triplica la carga de cada esfera? No uses números.
\espacio{2.5cm}
\begin{solucion}
$F' = k\dfrac{(3q_1)(3q_2)}{r^2} = 9\,k\dfrac{q_1 q_2}{r^2} = 9F$: la fuerza se hace nueve
veces mayor. La distancia no cambió.
\end{solucion}

% Ejercicio: electrostatica-11-047
\pregunta[1] Dos esferas se atraen con una fuerza determinada. ¿Qué sucede con la fuerza si la
carga de cada esfera se reduce a la tercera parte?
\espacio{2.5cm}
\begin{solucion}
$F' = k\dfrac{(q_1/3)(q_2/3)}{r^2} = \dfrac{F}{9}$: la fuerza se reduce a la novena parte.
\end{solucion}

% Ejercicio: electrostatica-11-055
\pregunta[1.5] Se tienen dos cargas de \qty{2}{\micro\coulomb} y \qty{8}{\micro\coulomb}
separadas \qty{10}{cm}. Calcula la fuerza, en newtons, que existe entre ellas y di si es de
atracción o de repulsión.
\espacio{4cm}
\begin{solucion}
$q_1 = \qty{2e-6}{C}$, $q_2 = \qty{8e-6}{C}$, $r = \qty{0,1}{m}$.

$F = \dfrac{(\num{8,99e9})(\num{2e-6})(\num{8e-6})}{(\num{0,1})^2} \approx \qty{14,4}{N}$.

Las dos cargas son positivas, así que es de \textbf{repulsión}. Ojo con el paso
$\qty{10}{cm} = \qty{0,1}{m}$: dejar los centímetros da un resultado diez mil veces menor.
\end{solucion}

% Ejercicio: electrostatica-11-054
\pregunta[1.5] Dos cargas de \qty{40}{\micro\coulomb} se repelen con una fuerza de
\qty{360}{N}. Calcula la distancia que las separa.
\espacio{4cm}
\begin{solucion}
Se despeja $r$ de $F = k\dfrac{q^2}{r^2}$:

$r = \sqrt{\dfrac{k q^2}{F}} =
\sqrt{\dfrac{(\num{8,99e9})(\num{40e-6})^2}{360}} \approx \qty{0,20}{m}$, es decir unos
\qty{20}{cm}.
\end{solucion}

\end{preguntas}
