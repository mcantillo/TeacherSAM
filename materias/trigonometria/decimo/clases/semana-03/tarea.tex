\begin{preguntas}

% Ejercicio: razones-trigonometricas-10-002, razones-trigonometricas-10-004, razones-trigonometricas-10-006
\pregunta Evalúa con la calculadora, con cuatro decimales:
\begin{opciones*}(3)
  \item $\sen \num{89,3}^\circ$
  \item $\tg \num{72,3}^\circ$
  \item $\cos \num{38,7}^\circ$
\end{opciones*}
\begin{solucion}[2cm]
a) $\sen \num{89,3}^\circ \approx \num{0,9999}$. \quad
b) $\tg \num{72,3}^\circ \approx \num{3,1334}$. \quad
c) $\cos \num{38,7}^\circ \approx \num{0,7804}$.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-008, razones-trigonometricas-10-010, razones-trigonometricas-10-012
\pregunta Encuentra el ángulo agudo $\theta$ (en grados, a la décima) en cada caso:
\begin{opciones*}(3)
  \item $\cos\theta = \num{0,9354}$
  \item $\cos\theta = \num{0,3535}$
  \item $\sen\theta = \num{0,7302}$
\end{opciones*}
\begin{solucion}[2cm]
a) $\theta \approx \num{20,7}^\circ$. \quad b) $\theta \approx \num{69,3}^\circ$. \quad
c) $\theta \approx \num{46,9}^\circ$.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-016
\pregunta En un triángulo rectángulo, el cateto adyacente a un ángulo de $40^\circ$ mide
$24$. Halla la longitud $x$ de la hipotenusa. Dibuja el triángulo y marca sus partes antes
de calcular.
\begin{solucion}[3cm]
El dato y la incógnita son el cateto adyacente y la hipotenusa: la razón que los contiene
es el coseno. $\cos 40^\circ = \dfrac{24}{x}$, así que
$x = \dfrac{24}{\cos 40^\circ} \approx \num{31,3}$.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-034
\pregunta Una escalera de $20$ pies de longitud se apoya en una pared y forma un ángulo de
$76^\circ$ con el suelo. ¿A qué altura de la pared llega el extremo superior de la
escalera?
\begin{solucion}[3cm]
La escalera es la hipotenusa y la altura es el cateto opuesto al ángulo de $76^\circ$:
$h = 20\sen 76^\circ \approx \num{19,4}$ pies.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-036
\pregunta Un alambre tensor atado a un poste forma un ángulo de $69^\circ$ con el suelo y
está fijado al suelo a $14$ pies del poste. ¿A qué altura del poste está atado el alambre?
\begin{solucion}[3cm]
La distancia al poste es el cateto adyacente y la altura el opuesto: la tangente.
$h = 14\tg 69^\circ \approx \num{36,5}$ pies.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-033
\pregunta Una trayectoria recta que sube una colina se eleva $26$ pies por cada $100$ pies
horizontales. ¿Qué ángulo forma con la horizontal?
\begin{solucion}[3cm]
Se conocen el cateto opuesto y el adyacente, así que se usa la tangente y se va de la
razón al ángulo: $\tg\theta = \dfrac{26}{100} = \num{0,26}$, luego
$\theta = \arctg \num{0,26} \approx \num{14,6}^\circ$.
\end{solucion}

\end{preguntas}
