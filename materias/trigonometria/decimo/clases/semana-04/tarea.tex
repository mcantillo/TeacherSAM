\begin{instrucciones}
Trabajo en casa, \textbf{con calculadora científica en modo grados} (\textsc{deg}). Antes de
empezar, comprueba: $\sen 30^\circ$ debe dar exactamente $0{,}5$; si no, tu calculadora está
en otro modo. Razones con \textbf{cuatro decimales}, ángulos con \textbf{una décima de grado}.
Total: \totalpuntos{} puntos.
\end{instrucciones}

\begin{preguntas}

% Ejercicio: razones-trigonometricas-10-003
\pregunta[0.5] Usa la calculadora (en modo grados) para evaluar $\tg \num{54,4}^\circ$. Da el
resultado con cuatro decimales.
\begin{solucion}[1.8cm]
$\tg \num{54,4}^\circ \approx \num{1,3968}$.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-005
\pregunta[0.5] Usa la calculadora (en modo grados) para evaluar $\cos \num{49,2}^\circ$. Da el
resultado con cuatro decimales.
\begin{solucion}[1.8cm]
$\cos \num{49,2}^\circ \approx \num{0,6534}$.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-040
\pregunta[1] Calcula con la calculadora $\num{24,6} \cos \num{74,3}^\circ$. Cuida el orden de
las teclas: primero el coseno del ángulo y después la multiplicación.
\begin{solucion}[2.5cm]
$\num{24,6} \cos \num{74,3}^\circ \approx \num{6,657}$.

Si da algo muy distinto, probablemente se calculó $\cos(\num{24,6} \times \num{74,3})$.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-042
\pregunta[1] Encuentra el ángulo agudo $\theta$ (en grados, a la décima) tal que
$\sen \theta = \num{0,6691}$.
\begin{solucion}[2.5cm]
$\theta = \sen^{-1}(\num{0,6691}) \approx \num{42,0}^\circ$.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-044
\pregunta[1] Encuentra el ángulo agudo $\theta$ (en grados, a la décima) tal que
$\tg \theta = \num{5,396}$.
\begin{solucion}[2.5cm]
$\theta = \tg^{-1}(\num{5,396}) \approx \num{79,5}^\circ$.

Tiene sentido: una tangente grande significa que el cateto opuesto es mucho mayor que el
adyacente, es decir, un ángulo cercano a $90^\circ$.
\end{solucion}

% Ejercicio: razones-trigonometricas-10-030
\pregunta[1] Resuelve el triángulo rectángulo con $\gamma = 90^\circ$ (hipotenusa $c$) en el
que $c = 41$ y $a = 40$. Dibújalo primero y marca sus partes.
\begin{solucion}[4cm]
$b = \sqrt{41^2 - 40^2} = \sqrt{1681 - 1600} = \sqrt{81} = 9$.

$\sen \alpha = \frac{40}{41}$, de donde $\alpha \approx \num{77,3}^\circ$, y
$\beta = 90^\circ - \alpha \approx \num{12,7}^\circ$.
\end{solucion}

\end{preguntas}
