\begin{preguntas}

\pregunta Escribe cada fracción como decimal y di si el decimal es finito o periódico:
\begin{opciones*}(4)
  \item $\dfrac{3}{8}$
  \item $\dfrac{5}{6}$
  \item $\dfrac{7}{11}$
  \item $\dfrac{2}{5}$
\end{opciones*}
\begin{solucion}[3cm]
a) $0{,}375$ (finito). b) $0{,}8\overline{3}$ (periódico). c) $0{,}\overline{63}$
(periódico). d) $0{,}4$ (finito).
\end{solucion}

\pregunta Escribe cada decimal como fracción irreducible:
\begin{opciones*}(4)
  \item $0{,}75$
  \item $0{,}\overline{6}$
  \item $0{,}\overline{27}$
  \item $2{,}\overline{1}$
\end{opciones*}
\begin{solucion}[4cm]
a) $\frac{3}{4}$. b) $10x - x = 6$, $x = \frac{6}{9} = \frac{2}{3}$. c) $100x - x = 27$,
$x = \frac{27}{99} = \frac{3}{11}$. d) $2 + \frac{1}{9} = \frac{19}{9}$.
\end{solucion}

\pregunta Ubica en la recta los números $\frac{1}{2}$, $-\frac{3}{4}$ y $\frac{5}{3}$.
\begin{center}
\begin{tikzpicture}[scale=1.6]
  \draw[->] (-2.4,0) -- (2.4,0);
  \foreach \x in {-2,...,2} \draw (\x,0.07) -- (\x,-0.07) node[below] {$\x$};
\end{tikzpicture}
\end{center}
\begin{solucion}
$-\frac{3}{4}$ entre $-1$ y $0$ (tres cuartos del camino hacia $-1$); $\frac{1}{2}$ en la
mitad entre $0$ y $1$; $\frac{5}{3} \approx 1{,}67$ entre $1$ y $2$.
\end{solucion}

\pregunta Divide $1 \div 7$ hasta que aparezca un patrón. ¿Por qué los decimales tienen que
repetirse? (Pista: ¿cuántos restos distintos puede dar una división entre 7?)
\renglones{3}
\begin{solucion}
$\frac{1}{7} = 0{,}\overline{142857}$. Al dividir entre 7 los restos solo pueden ser
$1, 2, \ldots, 6$ (si fuera 0, la división terminaría); en a lo más 6 pasos un resto se
repite y desde ahí se repiten las cifras.
\end{solucion}
