% Ejercicio: intervalos-002
\pregunta Escribe en forma de intervalo y representa en la recta real:
\begin{opciones*}(2)
  \item $x < -5$ y $x > -8$
  \item $x \ge 9$
  \item $x < 0$ y $x > -8$
  \item $x < -10$ y $x > 3$
\end{opciones*}
\espacio{5cm}
\begin{solucion}
a) $(-8, -5)$. b) $[9, \infty)$. c) $(-8, 0)$. d) $\varnothing$ (ningún número cumple las dos).
\end{solucion}

% Ejercicio: intervalos-009
% Ejercicio: intervalos-011
% Ejercicio: intervalos-013
% Ejercicio: intervalos-014
\pregunta Expresa cada intervalo como una desigualdad en términos de $x$:
\begin{opciones*}(4)
  \item $(-6, 8)$
  \item $[-2, 0)$
  \item $(-\infty, 3)$
  \item $(-\infty, \infty)$
\end{opciones*}
\espacio{2.5cm}
\begin{solucion}
a) $-6 < x < 8$. b) $-2 \le x < 0$. c) $x < 3$. d) $-\infty < x < \infty$ (todos los números
reales).
\end{solucion}

\end{preguntas}
